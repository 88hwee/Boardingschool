import React, { useState, useMemo } from 'react';
import { 
  Users, UserCheck, Activity, AlertCircle, 
  X, Calendar, AlertTriangle, CheckCircle2,
  Clock, BookOpen, MapPin, Search, UserX, ShieldAlert,
  ChevronDown, Plus
} from 'lucide-react';

// --- CONSTANTS & CONFIG ---
const ATTENDANCE_CATEGORIES = {
  PRESENT: ['Present', 'Late'],
  ACTIVITIES: ['CCA', 'Tuition (In-house)', 'Tuition (Outside)'],
  ABSENT: ['Absent with valid reason', 'Stay out', 'Truancy']
};

const ALL_STATUSES = [
  'Present', 'Late', 'CCA', 'Tuition (In-house)', 
  'Tuition (Outside)', 'Truancy', 'Absent with valid reason', 'Stay out'
];

const DISCIPLINE_OPTIONS = [
  'Cleanliness', 'Late for prep', 'Truancy', 
  'Broke curfew', 'Stealing', 'Bullying'
];

// Helper to get consistent colors for statuses
const getStatusConfig = (status) => {
  switch (status) {
    case 'Present': return { bg: 'bg-emerald-50', border: 'border-emerald-200', text: 'text-emerald-700', badge: 'bg-emerald-500', icon: <CheckCircle2 size={14}/> };
    case 'Late': return { bg: 'bg-yellow-50', border: 'border-yellow-300', text: 'text-yellow-700', badge: 'bg-yellow-500', icon: <Clock size={14}/> };
    case 'CCA': 
    case 'Tuition (In-house)': 
    case 'Tuition (Outside)': 
      return { bg: 'bg-blue-50', border: 'border-blue-200', text: 'text-blue-700', badge: 'bg-blue-500', icon: <Activity size={14}/> };
    case 'Absent with valid reason': return { bg: 'bg-slate-100', border: 'border-slate-300', text: 'text-slate-600', badge: 'bg-slate-400', icon: <UserX size={14}/> };
    case 'Stay out': return { bg: 'bg-orange-50', border: 'border-orange-200', text: 'text-orange-700', badge: 'bg-orange-500', icon: <MapPin size={14}/> };
    case 'Truancy': return { bg: 'bg-red-50', border: 'border-red-200', text: 'text-red-700', badge: 'bg-red-600', icon: <ShieldAlert size={14}/> };
    default: return { bg: 'bg-white', border: 'border-slate-200', text: 'text-slate-700', badge: 'bg-slate-300', icon: null };
  }
};

// --- MOCK DATA ---
const INITIAL_STUDENTS = [
  { id: '1', name: 'John Jacob Tejada', room: '532', bed: 'A', status: 'Present', discipline: [] },
  { id: '2', name: 'Jordan Reiner', room: '532', bed: 'B', status: 'Present', discipline: [{ id: 'd1', date: '21 Oct 2026, 10:00 AM', type: 'Late for prep', details: '15 minutes late for evening prep session.' }] },
  { id: '3', name: 'Ezra Lim Zhi En', room: '532', bed: 'C', status: 'Present', discipline: [] },
  { id: '4', name: 'Zack Anderson', room: '532', bed: 'D', status: 'Present', discipline: [] },
  
  { id: '5', name: 'Dion Maxson', room: '533', bed: 'A', status: 'Present', discipline: [] },
  { id: '6', name: 'Lionel Antonio', room: '533', bed: 'B', status: 'Present', discipline: [] },
  { id: '7', name: 'Jonathan Bryan', room: '533', bed: 'C', status: 'Present', discipline: [] },
  { id: '8', name: 'Marcellino Winatajaya', room: '533', bed: 'D', status: 'Present', discipline: [] },
  
  { id: '11', name: 'Zachary Loo', room: '534', bed: 'A', status: 'Present', discipline: [{ id: 'd3', date: '20 Oct 2026, 09:15 PM', type: 'Cleanliness', details: 'Bed unmade, trash left on desk.' }] },
  { id: '12', name: 'Dinh Huu Duc', room: '534', bed: 'B', status: 'Present', discipline: [] },
];

export default function BoardingDashboard() {
  const [students, setStudents] = useState(INITIAL_STUDENTS);
  const [searchQuery, setSearchQuery] = useState('');
  const [selectedStudentId, setSelectedStudentId] = useState(null);
  const [modalTab, setModalTab] = useState('attendance'); 
  const [disciplineDetails, setDisciplineDetails] = useState('');

  // --- METRICS ---
  const metrics = useMemo(() => {
    let presentCount = 0;
    let activitiesCount = 0;
    let absentCount = 0;

    students.forEach(s => {
      if (ATTENDANCE_CATEGORIES.PRESENT.includes(s.status)) presentCount++;
      else if (ATTENDANCE_CATEGORIES.ACTIVITIES.includes(s.status)) activitiesCount++;
      else if (ATTENDANCE_CATEGORIES.ABSENT.includes(s.status)) absentCount++;
    });

    return {
      total: students.length,
      present: presentCount,
      activities: activitiesCount,
      absent: absentCount
    };
  }, [students]);

  // --- GROUP BY ROOM ---
  const rooms = useMemo(() => {
    const grouped = {};
    students.forEach(s => {
      if (!grouped[s.room]) grouped[s.room] = [];
      grouped[s.room].push(s);
    });
    return grouped;
  }, [students]);

  // --- HANDLERS ---
  const handleMarkPresent = (studentId) => {
    setStudents(prev => prev.map(s => s.id === studentId ? { ...s, status: 'Present' } : s));
  };

  const handleUpdateStatus = (studentId, newStatus) => {
    setStudents(prev => prev.map(s => s.id === studentId ? { ...s, status: newStatus } : s));
    setSelectedStudentId(null); // Optional: auto-close modal on selection
  };

  const handleAddDiscipline = (studentId, type) => {
    const newIncident = {
      id: Date.now().toString(),
      date: new Date().toLocaleString('en-GB', { 
        day: '2-digit', month: 'short', year: 'numeric', 
        hour: '2-digit', minute: '2-digit', hour12: true 
      }),
      type,
      details: disciplineDetails.trim()
    };
    setStudents(prev => prev.map(s => 
      s.id === studentId ? { ...s, discipline: [newIncident, ...s.discipline] } : s
    ));
    setDisciplineDetails('');
  };

  const handleRemoveDiscipline = (studentId, incidentId) => {
    setStudents(prev => prev.map(s => 
      s.id === studentId ? { ...s, discipline: s.discipline.filter(d => d.id !== incidentId) } : s
    ));
  };

  const selectedStudent = students.find(s => s.id === selectedStudentId);

  return (
    <div className="min-h-screen bg-slate-50 font-sans text-slate-800 pb-16 sm:pb-0">
      
      {/* HEADER & METRICS */}
      <header className="bg-white border-b border-slate-200 sticky top-0 z-20 shadow-sm">
        <div className="max-w-7xl mx-auto px-4 py-4">
          <div className="flex flex-col sm:flex-row justify-between sm:items-center gap-4 mb-4">
            <div>
              <h1 className="text-xl sm:text-2xl font-bold text-slate-900 tracking-tight">Hall Operations</h1>
              <p className="text-xs sm:text-sm text-slate-500 font-medium mt-0.5">
                {new Date().toLocaleDateString('en-GB', { weekday: 'long', day: 'numeric', month: 'long', year: 'numeric' })}
              </p>
            </div>
            <div className="relative w-full sm:w-72">
              <Search className="absolute left-3 top-1/2 -translate-y-1/2 text-slate-400" size={18} />
              <input 
                type="text" 
                placeholder="Search..." 
                value={searchQuery}
                onChange={(e) => setSearchQuery(e.target.value)}
                className="w-full pl-10 pr-4 py-3 sm:py-2 bg-slate-100 border-none rounded-xl text-base focus:ring-2 focus:ring-indigo-500"
              />
            </div>
          </div>

          <div className="grid grid-cols-4 gap-2 sm:gap-4">
            <div className="bg-slate-50 rounded-xl p-2 sm:p-3 text-center sm:text-left sm:flex items-center gap-3">
              <div className="hidden sm:flex p-2 bg-slate-200 text-slate-600 rounded-lg"><Users size={20}/></div>
              <div>
                <p className="text-[10px] sm:text-xs font-semibold text-slate-500 uppercase tracking-wider">Total</p>
                <p className="text-lg sm:text-xl font-bold text-slate-800">{metrics.total}</p>
              </div>
            </div>
            <div className="bg-emerald-50 rounded-xl p-2 sm:p-3 text-center sm:text-left sm:flex items-center gap-3">
              <div className="hidden sm:flex p-2 bg-emerald-200 text-emerald-700 rounded-lg"><UserCheck size={20}/></div>
              <div>
                <p className="text-[10px] sm:text-xs font-semibold text-emerald-600 uppercase tracking-wider">Present</p>
                <p className="text-lg sm:text-xl font-bold text-emerald-700">{metrics.present}</p>
              </div>
            </div>
            <div className="bg-blue-50 rounded-xl p-2 sm:p-3 text-center sm:text-left sm:flex items-center gap-3">
              <div className="hidden sm:flex p-2 bg-blue-200 text-blue-700 rounded-lg"><Activity size={20}/></div>
              <div>
                <p className="text-[10px] sm:text-xs font-semibold text-blue-600 uppercase tracking-wider">Activity</p>
                <p className="text-lg sm:text-xl font-bold text-blue-700">{metrics.activities}</p>
              </div>
            </div>
            <div className="bg-red-50 rounded-xl p-2 sm:p-3 text-center sm:text-left sm:flex items-center gap-3">
              <div className="hidden sm:flex p-2 bg-red-200 text-red-700 rounded-lg"><AlertCircle size={20}/></div>
              <div>
                <p className="text-[10px] sm:text-xs font-semibold text-red-600 uppercase tracking-wider">Absent</p>
                <p className="text-lg sm:text-xl font-bold text-red-700">{metrics.absent}</p>
              </div>
            </div>
          </div>
        </div>
      </header>

      {/* MAIN ROOM LAYOUT */}
      <main className="max-w-7xl mx-auto px-4 py-6">
        <div className="space-y-6">
          {Object.entries(rooms).map(([roomNumber, roomStudents]) => {
            const matchesSearch = roomNumber.includes(searchQuery) || 
                                  roomStudents.some(s => s.name.toLowerCase().includes(searchQuery.toLowerCase()));
            if (!matchesSearch) return null;

            return (
              <div key={roomNumber} className="bg-white rounded-2xl border border-slate-200 shadow-sm overflow-hidden">
                <div className="bg-slate-100/50 px-4 py-2 border-b border-slate-200 flex items-center gap-2">
                  <BookOpen size={16} className="text-slate-500" />
                  <h2 className="text-sm font-bold text-slate-700">Room {roomNumber}</h2>
                </div>
                
                <div className="p-3 sm:p-4 grid grid-cols-2 lg:grid-cols-4 gap-3 sm:gap-4">
                  {roomStudents.map(student => {
                    const styleConfig = getStatusConfig(student.status);
                    const isPresent = student.status === 'Present';
                    
                    return (
                      <div key={student.id} className={`flex flex-col bg-white rounded-xl border-2 shadow-sm overflow-hidden transition-all ${styleConfig.border}`}>
                        
                        {/* TOP HALF: Avatar Area (Tap to mark present) */}
                        <button 
                          onClick={() => handleMarkPresent(student.id)}
                          className={`relative flex-1 p-4 flex flex-col items-center justify-center group active:bg-slate-50 transition-colors ${!isPresent ? 'bg-slate-50' : 'bg-white'}`}
                          title="Tap to mark Present"
                        >
                          <div className="absolute top-2 left-2 text-xs font-bold text-slate-400">{student.bed}</div>
                          
                          {student.discipline.length > 0 && (
                            <div className="absolute top-2 right-2 text-red-500"><AlertTriangle size={14} /></div>
                          )}

                          <div className={`w-14 h-14 rounded-full flex items-center justify-center text-xl font-bold mb-2 transition-transform group-active:scale-95
                            ${isPresent ? 'bg-emerald-100 text-emerald-700 border-2 border-emerald-200' : 'bg-slate-200 text-slate-500 border-2 border-slate-300'}`}
                          >
                            {student.name.charAt(0)}
                          </div>
                          
                          <h3 className="font-semibold text-slate-900 text-sm text-center leading-tight line-clamp-2 px-1">{student.name}</h3>
                          
                          {/* Hint overlay on non-present users */}
                          {!isPresent && (
                            <div className="mt-2 flex items-center gap-1 text-[10px] font-bold text-slate-400 bg-slate-200/50 px-2 py-1 rounded-full">
                              <CheckCircle2 size={12} /> Tap for Present
                            </div>
                          )}
                        </button>

                        {/* BOTTOM HALF: Status Badge (Tap to open modal) */}
                        <button
                          onClick={() => { 
                            setSelectedStudentId(student.id); 
                            setModalTab('attendance'); 
                            setDisciplineDetails(''); 
                          }}
                          className={`w-full py-2.5 px-3 flex items-center justify-between border-t active:brightness-95 transition-all
                            ${styleConfig.bg} ${styleConfig.border} ${styleConfig.text}`}
                        >
                          <div className="flex items-center gap-1.5 font-semibold text-xs truncate">
                            {styleConfig.icon}
                            <span className="truncate">{student.status}</span>
                          </div>
                          <ChevronDown size={14} className="opacity-70 shrink-0" />
                        </button>
                      </div>
                    );
                  })}
                </div>
              </div>
            );
          })}
        </div>
      </main>

      {/* BOTTOM SHEET MODAL */}
      {selectedStudent && (
        <>
          {/* Backdrop */}
          <div 
            className="fixed inset-0 bg-slate-900/40 backdrop-blur-sm z-40 transition-opacity"
            onClick={() => setSelectedStudentId(null)}
          />
          
          {/* Modal Content */}
          <div className="fixed inset-x-0 bottom-0 sm:inset-auto sm:top-1/2 sm:left-1/2 sm:-translate-x-1/2 sm:-translate-y-1/2 sm:w-[480px] bg-white rounded-t-3xl sm:rounded-2xl shadow-2xl z-50 flex flex-col max-h-[90vh] transition-transform transform translate-y-0">
            
            {/* Drag Handle (Mobile) */}
            <div className="w-full flex justify-center py-3 sm:hidden" onClick={() => setSelectedStudentId(null)}>
              <div className="w-12 h-1.5 bg-slate-300 rounded-full"></div>
            </div>

            {/* Header */}
            <div className="px-6 pb-4 sm:pt-6 flex justify-between items-start">
              <div className="flex items-center gap-4">
                <div className="w-12 h-12 rounded-full bg-indigo-100 text-indigo-700 flex items-center justify-center text-xl font-bold">
                  {selectedStudent.name.charAt(0)}
                </div>
                <div>
                  <h2 className="text-lg font-bold text-slate-900 leading-tight">{selectedStudent.name}</h2>
                  <p className="text-sm font-medium text-slate-500">Room {selectedStudent.room} • Bed {selectedStudent.bed}</p>
                </div>
              </div>
              <button onClick={() => setSelectedStudentId(null)} className="p-2 -mr-2 bg-slate-100 text-slate-500 hover:bg-slate-200 rounded-full">
                <X size={20} />
              </button>
            </div>

            {/* Tabs */}
            <div className="flex border-b border-slate-200 px-2">
              <button 
                onClick={() => setModalTab('attendance')}
                className={`flex-1 py-3 text-sm font-bold border-b-2 transition-colors ${
                  modalTab === 'attendance' ? 'border-indigo-600 text-indigo-700' : 'border-transparent text-slate-400'
                }`}
              >
                Set Status
              </button>
              <button 
                onClick={() => setModalTab('discipline')}
                className={`flex-1 py-3 text-sm font-bold border-b-2 flex items-center justify-center gap-2 transition-colors ${
                  modalTab === 'discipline' ? 'border-red-600 text-red-700' : 'border-transparent text-slate-400'
                }`}
              >
                Discipline
                {selectedStudent.discipline.length > 0 && (
                  <span className="bg-red-100 text-red-600 text-[10px] px-1.5 py-0.5 rounded-full">{selectedStudent.discipline.length}</span>
                )}
              </button>
            </div>

            {/* Scrollable Body */}
            <div className="p-6 overflow-y-auto overscroll-contain bg-slate-50/50 flex-1">
              
              {/* ATTENDANCE TAB */}
              {modalTab === 'attendance' && (
                <div className="space-y-3 pb-6">
                  {ALL_STATUSES.map(status => {
                    const isSelected = selectedStudent.status === status;
                    const config = getStatusConfig(status);
                    
                    return (
                      <button
                        key={status}
                        onClick={() => handleUpdateStatus(selectedStudent.id, status)}
                        className={`w-full flex items-center justify-between p-4 rounded-xl border-2 transition-all active:scale-[0.98] ${
                          isSelected ? `${config.bg} ${config.border} ring-2 ring-${config.badge.split('-')[1]}-500/20` : 'bg-white border-slate-200 hover:border-slate-300'
                        }`}
                      >
                        <div className="flex items-center gap-3">
                          <div className={`w-10 h-10 rounded-full flex items-center justify-center ${isSelected ? config.badge + ' text-white' : 'bg-slate-100 text-slate-500'}`}>
                            {config.icon || <UserCheck size={18}/>}
                          </div>
                          <span className={`text-base font-bold ${isSelected ? config.text : 'text-slate-700'}`}>{status}</span>
                        </div>
                        {isSelected && <CheckCircle2 className={config.text} size={24} />}
                      </button>
                    );
                  })}
                </div>
              )}

              {/* DISCIPLINE TAB */}
              {modalTab === 'discipline' && (
                <div className="space-y-6 pb-6">
                  <div>
                    <h3 className="text-xs font-bold text-slate-400 uppercase mb-3">Add Incident</h3>
                    <textarea
                      className="w-full mb-3 p-3 text-sm bg-white border border-slate-200 rounded-xl focus:ring-2 focus:ring-red-500 focus:border-red-500 outline-none resize-none transition-all placeholder:text-slate-400 shadow-sm"
                      rows="2"
                      placeholder="Optional details (e.g. 15 mins late, messy bed...)"
                      value={disciplineDetails}
                      onChange={(e) => setDisciplineDetails(e.target.value)}
                    />
                    <div className="flex flex-wrap gap-2">
                      {DISCIPLINE_OPTIONS.map(issue => (
                        <button
                          key={issue}
                          onClick={() => handleAddDiscipline(selectedStudent.id, issue)}
                          className="px-4 py-2.5 text-sm font-semibold rounded-xl border border-slate-200 bg-white text-slate-700 active:bg-red-50 active:text-red-700 active:border-red-200 shadow-sm flex items-center gap-2"
                        >
                          <Plus size={14} className="text-slate-400" /> {issue}
                        </button>
                      ))}
                    </div>
                  </div>

                  <div>
                    <h3 className="text-xs font-bold text-slate-400 uppercase mb-3">History</h3>
                    <div className="space-y-3">
                      {selectedStudent.discipline.length === 0 ? (
                        <div className="text-center py-6 bg-white border border-slate-200 rounded-xl">
                          <div className="bg-slate-100 w-12 h-12 rounded-full flex items-center justify-center mx-auto mb-2 text-slate-400">
                            <ShieldAlert size={20} />
                          </div>
                          <p className="text-sm font-semibold text-slate-600">No incidents recorded</p>
                          <p className="text-xs text-slate-400 mt-1">Good behavior!</p>
                        </div>
                      ) : (
                        selectedStudent.discipline.map(incident => (
                          <div key={incident.id} className="bg-white border border-slate-200 rounded-xl p-4 shadow-sm flex justify-between items-start">
                            <div className="flex gap-3 items-start">
                              <div className="bg-red-100 text-red-600 p-2 rounded-lg mt-0.5">
                                <AlertTriangle size={18} />
                              </div>
                              <div className="flex-1">
                                <p className="text-sm font-bold text-slate-800">{incident.type}</p>
                                <p className="text-xs font-medium text-slate-500 mt-0.5">{incident.date}</p>
                                {incident.details && (
                                  <p className="text-sm text-slate-600 mt-2 bg-slate-50 p-2.5 rounded-lg border border-slate-100 whitespace-pre-wrap">
                                    {incident.details}
                                  </p>
                                )}
                              </div>
                            </div>
                            <button 
                              onClick={() => handleRemoveDiscipline(selectedStudent.id, incident.id)}
                              className="p-2 text-slate-400 hover:text-red-600 active:bg-red-50 rounded-lg shrink-0 ml-2"
                            >
                              <X size={20} />
                            </button>
                          </div>
                        ))
                      )}
                    </div>
                  </div>
                </div>
              )}
            </div>
          </div>
        </>
      )}
    </div>
  );
}
